```cpp
// event_node.h (replaces BeginRun / EndRun)
protected:
    // Created -> Starting -> start_worker() -> Running. Any other state throws UsageError
    // ("Start() called twice" / "Start() after Stop(): create a new component").
    // If start_worker throws: state becomes Stopped, then rethrow.
    template <typename TStartWorker> void StartWith(TStartWorker&& start_worker);
    // Created -> Stopped; Running -> Stopping -> stop_worker() -> Stopped; Starting/Stopping:
    // wait (std::atomic::wait) until settled and retry; Stopped: return.
    // If stop_worker throws (self-stop): state goes back to Running, then rethrow.
    template <typename TStopWorker> void StopWith(TStopWorker&& stop_worker);

// observable.h additions (the state lives in the shared Wiring, so Subscriptions read it too)
    bool CompareExchangeState(RunState& expected, RunState desired);   // acq_rel CAS
    void WaitWhileState(RunState current) const;                       // state.wait(current)
    // SetState(s): release store + notify_all

// worker_thread.h
    // Stop(): throws UsageError when called on the worker thread itself (std::atomic<std::thread::id>
    // worker_id_ is stored by the worker thread as its first action and cleared on exit).
```
