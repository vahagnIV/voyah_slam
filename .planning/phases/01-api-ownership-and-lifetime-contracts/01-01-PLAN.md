```cpp
// libs/common/include/voyahslam/errors.h        guard VOYAHSLAM_ERRORS_H
namespace voyahslam {
class UsageError : public std::logic_error { public: using std::logic_error::logic_error; };
}

// libs/events/include/voyahslam/events/event_types.h   (EventType enum unchanged)
struct Event {
    EventType type;
    map::MapId map_id{};      // set by the producer before Notify; immutable afterwards
    bool priority = false;    // set by the producer before Notify; immutable afterwards
protected:
    explicit Event(EventType t);
    ~Event() = default;       // protected, non-virtual: events die through their shared_ptr control block
};
using EventPtr = std::shared_ptr<const Event>;

// inbox_core.h
class InboxCore {             // shared by AsyncObserver, its Listeners and producers' async entries
public:
    explicit InboxCore(std::string name);
    void Push(EventPtr event);                 // closed: drops the copy; else counts priority, enqueues, warns (D-24), notifies
    bool TryPop(EventPtr& out_event);          // decrements pending_priority for a priority event
    void Close();                              // ~AsyncObserver only: closed = true, then pop and release everything queued
    [[nodiscard]] bool closed() const;
    [[nodiscard]] uint32_t pending_priority() const;
    [[nodiscard]] size_t size_approx() const;
    [[nodiscard]] WakeSignal& wake_signal();
    [[nodiscard]] const std::string& name() const;
};

// listener.h — a copyable handle; no current_, no destructor work
class Listener { public: explicit Listener(std::shared_ptr<InboxCore> core); EventPtr GetNextEvent(); };

// async_observer.h — non-copyable, non-movable
class AsyncObserver {
public:
    explicit AsyncObserver(std::string name = "AsyncObserver");
    ~AsyncObserver();                                    // core_->Close()
    [[nodiscard]] Listener GetListener() const;
    bool TryPop(EventPtr& out_event);
    [[nodiscard]] uint32_t pending_priority() const;
    [[nodiscard]] size_t size_approx() const;
    [[nodiscard]] WakeSignal& wake_signal();
    [[nodiscard]] const std::string& name() const;
private:
    friend class Observable;
    std::shared_ptr<InboxCore> core_;
};

// subscription.h
enum class RunState : uint8_t { Created, Starting, Running, Stopping, Stopped };
[[nodiscard]] constexpr bool IsRunning(RunState s);     // Starting, Running or Stopping
struct Wiring {                                         // held by Observable and by every Subscription
    std::atomic<RunState> state{RunState::Created};
    uint64_t next_id = 1;
    struct AsyncEntry { uint64_t id; std::shared_ptr<InboxCore> core; };
    struct SyncEntry { uint64_t id; void (*fn)(void* ctx, const Event& event); void* ctx; };
    std::vector<AsyncEntry> async;                      // written only while !IsRunning(state)
    std::vector<SyncEntry> sync;
    void Remove(uint64_t id);
};
class [[nodiscard]] Subscription {                      // move-only
public:
    Subscription() = default;
    Subscription(Subscription&& other) noexcept;
    Subscription& operator=(Subscription&& other);      // disconnects the current one first (reset rules)
    ~Subscription();                                    // running producer: stderr + log + std::terminate (D-07); else Remove
    void reset();                                       // running producer: throws UsageError (D-07); idempotent
    [[nodiscard]] bool connected() const;
private:
    friend class Observable;
    Subscription(std::shared_ptr<Wiring> wiring, uint64_t id);
    std::shared_ptr<Wiring> wiring_;
    uint64_t id_ = 0;
};

// observable.h
class Observable {
public:
    Observable();                                           // makes the shared Wiring
    [[nodiscard]] Subscription ConnectAsync(AsyncObserver& observer);   // throws UsageError while running (D-05)
    [[nodiscard]] Subscription ConnectSync(void (*fn)(void* ctx, const Event& event), void* ctx);
    template <typename T> void Notify(std::shared_ptr<T>&& event);   // static_assert(std::is_base_of_v<Event, T>)
    [[nodiscard]] size_t async_count() const;
    [[nodiscard]] RunState state() const;
    void SetState(RunState state);                          // release store; called by EventNode only
};

// event_node.h
class EventNode {
public:
    explicit EventNode(std::string inbox_name = "EventNode");
    [[nodiscard]] Subscription AddObserver(AsyncObserver& observer);
    [[nodiscard]] Subscription AddSyncObserver(void (*handler)(void* ctx, const Event& event), void* ctx);
    [[nodiscard]] AsyncObserver& inbox();
    [[nodiscard]] size_t async_observer_count() const;
    [[nodiscard]] RunState run_state() const;
protected:
    [[nodiscard]] WakeSignal& wake_signal();   // inbox_.wake_signal()
    void BeginRun();                           // SetState(Running) before the worker thread is created
    void EndRun();                             // SetState(Stopped) after the worker thread is joined
    Observable observable_;
    AsyncObserver inbox_;
};
```

<assumption_delta_decision>
Detector signal: pluralization ("also", in the Phase 1 success criteria). The real transition this release makes is ownership going from one owner to several: an event was owned by its producer's pool (singular) and is now co-owned by every holder; a connection was a raw pointer owned by nobody and is now a Subscription co-owning the Wiring block with its producer.
- Primary noun: the event itself (a self-owning `std::shared_ptr<const T>`) and the `Subscription` (the connection's identity).
- Decision: promote. The pool is deleted (D-02, plan 01-05), not kept alongside; the raw-pointer registration and `SetWakeSignal` are removed, not kept alongside.
- Invariant test (accepted, Task 2): `DestructionOrderTest` (every destruction order frees every event once) goes red if any later change reintroduces a single owner that can die first.
</assumption_delta_decision>
