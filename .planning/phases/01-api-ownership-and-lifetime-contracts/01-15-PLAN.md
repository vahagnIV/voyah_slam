```cpp
class OrbFeatureExtractor {
public:
    // Per-thread scratch for DetectAndExtract (the image pyramid). Not thread-safe: one per
    // extraction thread (stereo: two if left and right are extracted in parallel). Reusing it
    // across frames avoids per-frame allocation; it re-initialises when the image size or the
    // extractor's level count changes.
    class Workspace {
    public:
        Workspace() = default;
    private:
        friend class OrbFeatureExtractor;
        orb::ImagePyramid pyramid_;
    };
    [[nodiscard]] TFeatures DetectAndExtract(const images::TWorkingImage& image, size_t number_of_features,
                                             Workspace& out_workspace) const;
    // Convenience: builds a temporary Workspace per call (allocates); fine for tests and one-offs.
    [[nodiscard]] TFeatures DetectAndExtract(const images::TWorkingImage& image, size_t number_of_features) const;
    // private helpers take const orb::ImagePyramid& pyramid
};
// orb::ImagePyramid: const operator[]; Width(), Height() of level 0; Initialize() clears before building;
// members default-initialised.
```
