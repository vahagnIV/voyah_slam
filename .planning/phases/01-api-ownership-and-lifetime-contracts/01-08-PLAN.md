```cpp
// libs/slam/include/voyahslam/mapping/map_snapshot.h   guard VOYAHSLAM_MAPPING_MAP_SNAPSHOT_H, namespace voyahslam::mapping
struct MapSnapshot {                      // plain data; no pointer into any replica
    struct Map {
        map::MapId id;
        uint32_t epoch = 0;
        std::vector<map::MapPointId> point_ids;            // alive points, ascending
        std::vector<geometry::TPoint3DF> point_positions;  // world, same order
        std::vector<map::FrameId> frame_ids;               // ascending
        std::vector<geometry::RigidTransformation> frame_poses;  // T_cw, same order
    };
    std::vector<Map> maps;                // every live map, ascending id
    std::optional<map::MapId> active_map;
    size_t num_transactions_applied = 0;
    [[nodiscard]] const Map* Find(map::MapId id) const;   // nullptr if not live
};

// libs/slam/include/voyahslam/mapping/map_follower.h   header-only template, namespace voyahslam::mapping
template <typename TFeatureExtractor>
class MapFollower {
public:
    MapFollower();                                         // inbox named "MapFollower"
    MapFollower(const MapFollower&) = delete;
    MapFollower& operator=(const MapFollower&) = delete;
    // Subscribe before the Mapper starts: Subscription s = mapper.AddObserver(follower.inbox());
    [[nodiscard]] events::AsyncObserver& inbox();
    // Applies every pending MapTransaction on the calling thread, in order; other events are released.
    // Returns the number of transactions applied by this call.
    size_t Update();
    // Update(), atlas() and Snapshot() belong to one thread (asserted in debug from the first Update()).
    [[nodiscard]] const tracking::AtlasReplica<TFeatureExtractor>& atlas() const;
    [[nodiscard]] std::shared_ptr<const MapSnapshot> Snapshot() const;   // O(map); any thread may hold the result
    [[nodiscard]] size_t num_transactions_applied() const;
};
```
